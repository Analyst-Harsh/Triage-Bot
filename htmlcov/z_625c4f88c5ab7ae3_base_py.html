15class TriageNode(ABC): 

16 """Template-method contract every graph node implements. 

17 

18 Concrete subclasses implement `execute()` with node-specific logic, 

19 constructing/validating the relevant Pydantic model(s) and writing it 

20 into the returned partial update (nodes construct/validate the Pydantic 

21 model, then write it into the TypedDict slot). 

22 

23 `__call__` is the thin, uniform seam LangGraph actually invokes: it 

24 checks the run's cost/iteration budget (`check_budget`, 

25 `graph/nodes/utils/budget_guard.py`) before `execute()` is allowed to 

26 start, bumps the `run_meta.iteration_count` guardrail counter after a 

27 successful `execute()`, and wraps the call in a Langfuse span 

28 (`observability.tracing.node_span`) enriched with this node's own 

29 duration/cost accounting — the cross-cutting concerns every node needs, 

30 applied once here rather than per subclass. It also marks that span 

31 `level="ERROR"` whenever `execute()` returns a `run_meta` with new 

32 entries in `errors` (via `RunMeta.with_error`) — the soft-failure 

33 counterpart to `handle_node_error` below, for a node that catches and 

34 converts a real failure into data (e.g. `AutoPostNode`/`ApprovalQueueNode` 

35 on a GitHub post failure) instead of raising. 

36 

37 Error handling for an uncaught exception is deliberately NOT done here. 

38 It's owned by LangGraph's native per-node `error_handler` hook, wired 

39 once, graph-wide, via `StateGraph.set_node_defaults(error_handler=...)` 

40 in `graph/builder.py` — that's LangGraph's own tested mechanism for 

41 "raise in execute() -> RunError + status=FAILED", so it isn't duplicated 

42 here. 

43 

44 A future subgraph-backed node (e.g. Researcher's tool-calling loop) does 

45 NOT go through this class. It's registered as its own compiled 

46 `StateGraph(TriageState)` passed directly to `add_node()`, so LangGraph's 

47 automatic subgraph detection (checkpoint namespacing, nested streaming) 

48 applies — wrapping `.invoke()` inside `execute()` would silently defeat 

49 that detection, since it relies on inspecting closures/Pregel instances, 

50 not attribute access through `self`. 

51 """ 

52 

53 name: ClassVar[NodeName] 

54 """Canonical graph node name — used as the add_node() key and as 

55 RunError.node_name on failure."""

57 async def __call__(self, state: TriageState) -> TriageStateUpdate: 

58 run_meta = state["run_meta"] 

59 with structlog.contextvars.bound_contextvars( 

60 run_id=str(run_meta.run_id), 

61 thread_id=run_meta.thread_id, 

62 trace_id=run_meta.trace_id, 

63 ): 

64 log.info("node_started", node=self.name) 

65 started_at = time.monotonic() 

66 async with node_span(self.name) as span: 

67 check_budget(run_meta, node_name=self.name) 

68 update = await self.execute(state) 

69 duration_ms = round((time.monotonic() - started_at) * 1000, 2) 

70 base_run_meta = update.get("run_meta", state["run_meta"]) 

71 update["run_meta"] = base_run_meta.with_usage(iterations=1) 

72 if span is not None: 

73 # A node can append a RunError without raising (e.g. 

74 # AutoPostNode/ApprovalQueueNode on a real GitHub post 

75 # failure -- see RunMeta.with_error) -- new_errors 

76 # generalizes that signal into the span for any node 

77 # using the same pattern, not just these two. 

78 new_errors = update["run_meta"].errors[len(run_meta.errors) :] 

79 span.update( 

80 metadata={ 

81 "duration_ms": duration_ms, 

82 "authoritative_cost_usd_delta": ( 

83 update["run_meta"].estimated_cost_usd - run_meta.estimated_cost_usd 

84 ), 

85 "authoritative_cache_read_tokens_delta": ( 

86 update["run_meta"].cache_read_tokens - run_meta.cache_read_tokens 

87 ), 

88 **({"new_error_count": len(new_errors)} if new_errors else {}), 

89 }, 

90 level="ERROR" if new_errors else None, 

91 status_message=( 

92 "; ".join(e.error_message for e in new_errors) if new_errors else None 

93 ), 

94 ) 

95 log.info("node_finished", node=self.name, duration_ms=duration_ms) 

96 

97 return update
