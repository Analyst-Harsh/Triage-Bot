1"""Langfuse tracing: the OpenTelemetry + Langfuse half of observability/'s 

2cross-cutting systems (see docs/summary.md's "Observability" section) -- 

3`logging_config.py` already covers structured logs; this module covers the 

4trace/span side, degrading to a no-op when unconfigured, the same contract 

5`utils/episodic_memory_store.py`'s `NullEpisodicMemoryStore` uses. 

6 

7A single `langfuse.langchain.CallbackHandler`, passed into the top-level 

8`graph.ainvoke()` call in `main.py`, is what produces the entire nested span 

9tree for free: LangGraph propagates that config's callbacks into every node, 

10every nested subgraph (the Researcher's/Drafter's own compiled 

11`StateGraph`s), and every LLM/tool call inside them via LangChain's own 

12contextvar-based config propagation -- no per-node code is required for that 

13structural nesting. What this module adds on top: 

14 

15- `root_span`: one root span for `main()`'s whole run, bound to a 

16 deterministic trace ID (`create_trace_id`, seeded from `thread_id`) so the 

17 initial `ainvoke()` and a later interrupt/resume share one trace even 

18 though they're two separate calls. 

19- `node_span`: creates a span for one node's body (`TriageNode.__call__`, 

20 `AgentSubgraph.assemble_node`) so it can enrich its own span with this 

21 repo's own cost/token accounting, namespaced `authoritative_*` wherever 

22 it's attached -- deliberately distinct from Langfuse's own per-generation 

23 cost estimate (derived from Langfuse's own model price table, not this 

24 repo's litellm-based, prompt-cache-aware `llm/pricing.py::estimate_cost_usd`, 

25 which is what `RunMeta`'s guardrail actually enforces against -- see 

26 `docs/agent/architecture-conventions.md`). 

27 

28 This creates its own span rather than enriching whatever span the 

29 `CallbackHandler` already auto-created for the same node dispatch, even 

30 though that seems redundant at first (a first cut of this tried exactly 

31 that "ambient current span" approach via `Langfuse.update_current_span`, 

32 and it silently attached metadata to the *wrong* span every time -- 

33 confirmed empirically, not just in theory: LangChain's async callback 

34 dispatch (`langchain_core/callbacks/manager.py::ahandle_event`) fires each 

35 handler's `on_chain_start` via `asyncio.gather`, so the Langfuse handler's 

36 `context.attach()` -- the call that makes its new span "current" -- runs 

37 inside an isolated `asyncio.create_task`. Per Python's `contextvars` 

38 semantics, a mutation made inside a spawned task's context never 

39 propagates back to the caller once that task completes, so by the time 

40 `RunnableCallable.ainvoke()` (`langgraph/_internal/_runnable.py`) resumes 

41 and actually calls our node body, "current" still points at whatever was 

42 ambient *before* the callback dispatch ran -- the parent span, not the 

43 just-created per-node one. Holding a span object directly, created and 

44 updated within our own single coroutine, sidesteps this entirely: nothing 

45 about it depends on inheriting context across that task boundary). 

46 

47Never pass a `SecretStr` object (or its resolved value) into a span/trace 

48attribute -- only into the `Langfuse` client constructor itself (see 

49`docs/agent/security.md`). 

50"""

64def ensure_langfuse_client(settings: Settings) -> None: 

65 """Idempotent, process-wide setup -- mirrors `logging_config.configure_logging()`'s 

66 `_configured` guard. Constructs the one `Langfuse` client this process 

67 uses (retrievable anywhere afterward via `langfuse.get_client()`, which 

68 resolves to it automatically since it's the only client instantiated) 

69 if both `settings.langfuse_public_key`/`langfuse_secret_key` are set; 

70 otherwise does nothing; every other function in this module then stays a 

71 no-op, exactly like the unset-`episodic_memory_database_url` case 

72 degrades to `NullEpisodicMemoryStore`. 

73 """ 

74 global _configured 

75 if _configured: 

76 return 

77 if settings.langfuse_public_key is None or settings.langfuse_secret_key is None: 

78 return 

79 Langfuse( 

80 public_key=settings.langfuse_public_key.get_secret_value(), 

81 secret_key=settings.langfuse_secret_key.get_secret_value(), 

82 base_url=settings.langfuse_host, 

83 ) 

84 _configured = True

87def create_trace_id(seed: str) -> str: 

88 """Deterministic 32-hex-char trace ID derived from `seed` (pure hashing -- 

89 no client/network involved, safe to call even when tracing isn't 

90 configured). Called with `thread_id` (repo#issue_number, already 

91 deterministic) so `RunMeta.trace_id` is always populated the same way for 

92 a given thread, ready for a future out-of-process resume to continue the 

93 same trace by re-deriving the identical ID from the same seed.""" 

94 return Langfuse.create_trace_id(seed=seed)

97def build_callback_handler(trace_id: str | None = None) -> CallbackHandler | None: 

98 """The one `CallbackHandler` instance `main.py` passes into every 

99 `graph.ainvoke()` call for a run -- `None` when tracing isn't configured, 

100 so callers pass `callbacks=[handler] if handler else []`. 

101 

102 `trace_id` defaults to `None` -- the handler is constructed with no 

103 `trace_context`, which every current caller relies on: the handler's own 

104 `_take_root_trace_context` always honors an explicit `trace_context` over 

105 whatever's ambiently open in OTel context, which (verified against the 

106 installed SDK) produced a second, sibling root-level span in the same 

107 trace when both `root_span` and this handler were given the same 

108 `trace_id` explicitly -- two independent root observations tagged with 

109 the same trace ID, neither referencing the other as parent. With no 

110 `trace_context`, this handler's own root chain span instead falls 

111 through to normal ambient nesting and attaches under whatever 

112 `root_span` already opened, which is always already active by the time 

113 `graph.ainvoke()` fires its first callback, since `root_span` wraps that 

114 call -- this is `main.py`'s path today. 

115 

116 Pass an explicit `trace_id` only when there is no ambient `root_span` 

117 open to nest under -- e.g. a future FastAPI resume-only handler that 

118 recovers `trace_id` from a checkpointed `RunMeta` and fires a single 

119 `ainvoke(Command(resume=...))` with nothing wrapping it. There, ambient 

120 context has nothing to offer, so binding via `trace_context` explicitly 

121 is the only way to keep the resume in the same trace as the original 

122 run -- the exact case this parameter exists for, not yet exercised by 

123 any caller in this codebase. 

124 """ 

125 if not _configured: 

126 return None 

127 trace_context: TraceContext | None = {"trace_id": trace_id} if trace_id is not None else None 

128 return CallbackHandler(trace_context=trace_context)

131@asynccontextmanager 

132async def root_span( 

133 *, name: str, trace_id: str, session_id: str, metadata: Mapping[str, object] 

134) -> AsyncGenerator[None]: 

135 """Wraps `main()`'s whole run in one Langfuse span bound to `trace_id`, 

136 with `session_id`/`metadata` attached at the trace level -- everything 

137 nested inside (both `graph.ainvoke()` calls, across the interrupt/resume 

138 boundary, plus the issue fetch and result-file write around them) shares 

139 this one trace. `session_id=thread_id` also groups the initial run and 

140 its resume together in Langfuse's own UI. 

141 

142 An `@asynccontextmanager` wrapper around the SDK's own context manager, 

143 which is sync-only (`start_as_current_observation` returns an 

144 `_AgnosticContextManager` supporting `__enter__`/`__exit__` only, not the 

145 async dunder methods) -- a plain `with` block nests fine inside an async 

146 generator function, since nothing in the SDK's enter/exit does blocking 

147 I/O. 

148 

149 A no-op `nullcontext`-equivalent when tracing isn't configured. 

150 """ 

151 if not _configured: 

152 yield 

153 return 

154 client = get_client() 

155 with ( 

156 client.start_as_current_observation( 

157 name=name, as_type="span", trace_context={"trace_id": trace_id} 

158 ), 

159 propagate_attributes(session_id=session_id, metadata=dict(metadata)), 

160 ): 

161 try: 

162 yield 

163 finally: 

164 # Buffered client-side, same as any OTel exporter -- main.py is a 

165 # short-lived script, so an explicit flush here (rather than 

166 # waiting for the background flush interval) is what guarantees 

167 # spans actually reach Langfuse before the process exits. 

168 client.flush() 

169

171@asynccontextmanager 

172async def node_span(name: str) -> AsyncGenerator[LangfuseSpan | None]: 

173 """Wraps one graph node's body in its own span, named `name` (`self.name` 

174 at each of this function's two call sites -- `TriageNode.__call__`, 

175 `AgentSubgraph.assemble_node` -- since the span name is only known 

176 per-instance, not at class-definition time, which is also why this is a 

177 plain wrapped function rather than Langfuse's own `@observe()` decorator). 

178 

179 Yields the span itself so the caller can `span.update(metadata=...)` 

180 with its own already-computed numbers before the block exits -- see this 

181 module's docstring for why this creates its own span rather than 

182 enriching the `CallbackHandler`'s auto-created one for the same node 

183 dispatch. Yields `None` when tracing isn't configured -- callers must 

184 guard on that before calling `.update()`. 

185 """ 

186 if not _configured: 

187 yield None 

188 return 

189 with get_client().start_as_current_observation(name=name, as_type="span") as span: 

190 yield span
