30class SpamCloseNode(TriageNode): 

31 """Handles issues the Planner classified as `SPAM_OR_ABUSE` 

32 (see `route_after_planner`, `graph/nodes/routing.py`): builds a 

33 one-action `close` draft directly -- no Researcher/Drafter LLM call 

34 needed, since the action is fully determined by the Planner's own 

35 classification -- and hardcodes its risk to `RiskLevel.HIGH` by policy, 

36 exactly like `RiskCheckNode` does for `label`/`code_fix` (see that 

37 node's docstring): a spam call is a Planner judgment that can be wrong, 

38 so closing a real user's issue on the strength of it always requires a 

39 human to confirm first, never an automatic post. 

40 

41 Skips `RiskCheckNode`/`AutoPostNode` entirely and routes straight to 

42 `ApprovalQueueNode` (see `graph/builder.py`), which is the single place 

43 that actually posts the close (on approval) and writes the completed 

44 episode to memory -- this node does neither itself, unlike the old 

45 `SpamRejectedNode` it replaces. 

46 """ 

47 

48 name: ClassVar[NodeName] = NodeName.SPAM_CLOSE 

49 

50 async def execute(self, state: TriageState) -> TriageStateUpdate: 

51 planner_output = state["planner_output"] 

52 if planner_output is None: 

53 raise ValueError("spam_close called before planner_output was set") 

54 

55 issue = state["issue"] 

56 log.info( 

57 "issue_flagged_as_spam", 

58 issue_number=issue.issue_number, 

59 reasoning=planner_output.reasoning, 

60 ) 

61 

62 draft = DraftOutput( 

63 actions=[ 

64 DraftedAction( 

65 action=CloseAction( 

66 reason="spam or abuse", 

67 close_comment=_SPAM_CLOSE_COMMENT, 

68 ), 

69 rationale=planner_output.reasoning, 

70 ) 

71 ], 

72 overall_rationale=planner_output.reasoning, 

73 unsupported_claims=[], 

74 drafted_at=datetime.now(UTC), 

75 ) 

76 risk_assessment = RiskAssessment( 

77 action_assessments=[ 

78 ActionRiskAssessment( 

79 level=RiskLevel.HIGH, 

80 risk_factors=["spam_or_abuse close always requires human review by policy"], 

81 reasoning=( 

82 "Closing an issue on the strength of a spam/abuse classification " 

83 "always requires human review by policy, regardless of the " 

84 "Planner's confidence -- a false positive here silently closes a " 

85 "real user's issue." 

86 ), 

87 ) 

88 ],

90 )

92 action_results=[ActionPostResult(outcome=PostOutcome.QUEUED)], 

93 evaluated_at=datetime.now(UTC), 

94 ) 

95 

96 return TriageStateUpdate( 

97 draft=draft, 

98 risk_assessment=risk_assessment, 

99 post_results=post_results, 

100 status=RunStatus.PENDING_APPROVAL, 

101 )
