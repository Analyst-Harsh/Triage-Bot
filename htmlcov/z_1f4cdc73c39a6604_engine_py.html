1"""SQLAlchemy async engine + session factory for the ORM-backed 

2`triage_runs` table (`repositories.triage_run_repository`). Separate from 

3the checkpointer's own `psycopg_pool.AsyncConnectionPool` 

4(`utils.postgres_pool`) -- `langgraph-checkpoint-postgres` isn't built on 

5SQLAlchemy, so the two can't share a pool object; both point at the same 

6Postgres instance via the same `settings.database_url`. Three connection 

7pools, one database: the checkpointer's `psycopg_pool`, this SQLAlchemy 

8engine, and `utils/episodic_memory_store.py`'s own `AsyncPostgresStore` pool 

9-- the natural consequence of putting a real ORM in front of this one table 

10while the checkpointer and episodic memory store each stay on their own 

11native driver. This engine's pool is explicitly sized (`pool_size=5, 

12max_overflow=10`, capped at 15 connections) rather than left at whatever 

13SQLAlchemy's own default happens to be, so the total budget across all three 

14pools is a deliberate, stated number: this engine's 15 + the checkpointer 

15pool's `max_size=10` (`utils/postgres_pool.py`) + the episodic store's 

16`max_size=5` = 30, comfortably under Postgres's default 

17`max_connections=100` -- assuming a single API process/replica; this budget 

18must be revisited (lower per-process caps, or an external pooler) before 

19running more than one. 

20""" 

21 

22from collections.abc import AsyncGenerator 

23from contextlib import asynccontextmanager 

24 

25from sqlalchemy.ext.asyncio import ( 

26 AsyncEngine, 

27 AsyncSession, 

28 async_sessionmaker, 

29 create_async_engine, 

30)

35def _sqlalchemy_url(database_url: str) -> str: 

36 """SQLAlchemy needs the `+psycopg` dialect suffix to select the 

37 async-capable psycopg v3 dialect; every other consumer of 

38 `settings.database_url` in this codebase (the checkpointer's 

39 `psycopg_pool`, `AsyncPostgresStore`) takes the plain `postgresql://` 

40 DSN directly. Converting it in this one place keeps that distinction 

41 from leaking into callers.""" 

42 return database_url.replace("postgresql://", "postgresql+psycopg://", 1) 

43 

44 

45@asynccontextmanager 

46async def triage_run_engine(database_url: str) -> AsyncGenerator[AsyncEngine]: 

47 """Opens one engine for the process's lifetime, creates the 

48 `triage_runs` table if it doesn't exist yet (idempotent, safe on every 

49 process start -- the ORM-native equivalent of `utils/episodic_memory_ 

50 store.py`'s `pg_store.setup()` convention), and disposes the engine's 

51 pool on exit. `create_all` is a sync method, hence `run_sync` to bridge 

52 it through the async connection. `pool_pre_ping=True` brings this engine 

53 to parity with the other two pools, which get idle/lifetime connection 

54 recycling for free from `psycopg_pool`'s own defaults -- the 

55 `triage_runs` table this pool backs can go untouched for hours/days 

56 while a row sits at `pending_approval`, making a dead connection on the 

57 next claim attempt a real risk without it.""" 

58 engine = create_async_engine( 

59 _sqlalchemy_url(database_url), pool_size=5, max_overflow=10, pool_pre_ping=True 

60 ) 

61 async with engine.begin() as conn: 

62 await conn.run_sync(Base.metadata.create_all) 

63 try: 

64 yield engine 

65 finally: 

66 await engine.dispose()

70 return async_sessionmaker(engine, expire_on_commit=False)
